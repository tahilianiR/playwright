### Approach 1: split('').reverse().join('')

```ts
function reverseString(str: string): string {
  return str.split('').reverse().join('');
}

console.log(reverseString('playwright')); // thgirwyalp
```

### Approach 2: normal for loop (reverse and forward)

reverse loop 

```ts
console.log(reverseString("Hello"))

function reverseString(str: string): string {
  let result = '';
  for (let i = str.length - 1; i >= 0; i--) {
    result += str[i];
  }
  return result;
}
```

loop from start will also work 
```ts
console.log(reverseString("Hello"))

function reverseString(a:string) {
    let rev = ""
   for(let i=0; i< a.length; i++) {
    rev = a[i] + rev
   }
   return rev
}

```


### Approach 3: for...of loop (adding each character to the front)

```ts
console.log(reverseString("Hello"))

function reverseString(a:string) {
    let rev = ""
    for (const ch of a) {
    rev = ch + rev;
    }
   return rev
}
```

### Notes
- Time complexity is O(n). Strings are immutable in JS, so a new string is always built.
- An empty string returns an empty string.
- `split('')` breaks emoji into broken halves. Use `[...str].reverse().join('')` if emoji must work.
