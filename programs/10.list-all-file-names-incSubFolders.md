import { test } from 'playwright/test'
import * as fs from 'fs'
import * as path from 'path'

test.describe('Test123', () => {
    test('list files', async () => {
        listFiles("C:/Users/tahiliani_r/Documents")
    })
})

function listFiles(dir: string) {

    const entries = fs.readdirSync(dir, { withFileTypes: true });
    for (const entry of entries) {
        const fullPath = path.join(dir, entry.name);
        if (entry.isDirectory()) {
            listFiles(fullPath); // Recursively list files in subdirectories
        } else {
            console.log(fullPath); // Print the file path
        }
    }

}

//if readdirSync called without { withFileTypes: true }. Without that option it returns plain strings (file names), and strings don’t  have isDirectory().
