### Approach 1: using mathematics 

```ts
function reverseNumber(num: number): number {
  const sign = num < 0 ? -1 : 1;
  num = Math.abs(num);
  let rev = 0;
  while (num > 0) {
    rev = rev * 10 + (num % 10); // take last digit, append to rev
    num = Math.floor(num / 10);  // drop last digit
  }
  return sign * rev;
}

console.log(reverseNumber(12345)); // 54321
console.log(reverseNumber(-120));  // -21
```

### Approach 2:  toString().split('').reverse().join('') -> Number()

```ts
function reverseNumber(num: number): number {
  const rev = Number(Math.abs(num).toString().split('').reverse().join(''));
  return Math.sign(num) * rev;
}
```

### Notes
- Trailing zeros disappear, because `120` reverses to `021`, which is `21` as a number.
- Keep the sign separate, or negative numbers break.
- Time complexity is O(d), where d is the number of digits.
