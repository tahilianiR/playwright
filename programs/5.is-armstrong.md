function isArmstrong(num: number): boolean {
  let digitsArray : number[] = num.toString().split('').map(c => Number(c))  //can also use : .map(Number)
  
  const len = digitsArray.length;
  let total = 0;
  for(const digit of digitsArray) {
    total = total + digit ** len;                                            // Math.pow(digit,len)
  }

  return total === num
}

console.log(isArmstrong(153));   // true  → 1³ + 5³ + 3³ = 153
console.log(isArmstrong(9474));  // true  → 9⁴ + 4⁴ + 7⁴ + 4⁴ = 9474
console.log(isArmstrong(123));   // false → 1³ + 2³ + 3³ = 36
