Anagram check - two strings are anagrams if they contain the same letters in the same counts, in any order.

### Approach 1: sort both strings

```ts
function isAnagram(s1: string,s2 : string): boolean {
  
  const s1_sorted = s1.replace(/\s/g,'').toLowerCase().split('').sort().join('') 
  const s2_sorted = s2.replace(/\s/g,'').toLowerCase().split('').sort().join('') 
  return s1_sorted == s2_sorted
}

console.log(isAnagram('Listen', 'Silent')); // true
console.log(isAnagram('Dormitory', 'Dirty room')); // true
console.log(isAnagram('Hello', 'World')); // false
console.log(isAnagram('peek', 'keep')); // true
console.log(isAnagram('expect', 'except')); // true
console.log(isAnagram('mother in law', 'hitler woman')); // true


//To remove all whitespace, including tabs and line breaks, use \s
//  "/ ... /" marks a regular expression, which is a search pattern.
//g applies the replacement to every match. Without it, only the first space is removed.

```

### Approach 2: count characters - Create Map 

```ts
function buildCharMap(str: string): Map<string, number> {
  const map = new Map<string, number>();
  for (const ch of str.toLowerCase().replace(/\s/g, '')) {
    map.set(ch, (map.get(ch) ?? 0) + 1);
  }
  return map;
}

function areMapsEqual(map1: Map<string, number>, map2: Map<string, number>): boolean {
  if (map1.size !== map2.size) return false;   // different number of distinct characters

  for (const [ch, count] of map1) {
    if (map2.get(ch) !== count) return false;  // character missing or count differs
  }
  return true;
}

function isAnagram(s1: string, s2: string): boolean {
  return areMapsEqual(buildCharMap(s1), buildCharMap(s2));
}

console.log(isAnagram('Listen', 'Silent'));         // true
console.log(isAnagram('Dormitory', 'Dirty room'));  // true
console.log(isAnagram('Hello', 'World'));           // false
console.log(isAnagram('aab', 'abb'));               // false

//What each map looks like for 'Hello' and 'olleh':
//map1: h → 1, e → 1, l → 2, o → 1
//map2: o → 1, l → 2, e → 1, h → 1
```

### Notes
- Sorting is O(n log n). Counting is O(n), so it is faster for long strings.
- Decide up front whether case and spaces matter. The code above ignores both.
- Different lengths can never be anagrams, so check that first.
