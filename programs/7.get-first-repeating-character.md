function getFirstRepeatingChar(str :string) : string {

    const map = new Map<string,number>();			// const list = new Set<string>();

    for(const ch of str) {
        if (map.has(ch)) return ch;
        map.set(ch,1)								// list.add(ch)
    }

    return "notfound"
}
