```ts
const arr = [2,1,4,3]

const evenArr = []
const oddArr = []
for(const digit of arr) {
  if(digit%2 == 0) {
    evenArr.push(digit)
  } else {
    oddArr.push(digit)
  }
}

console.log(evenArr)
console.log(oddArr)
console.log(evenArr.concat(oddArr));
```

```ts
const arr = [2,1,4,3];

const evenArr = arr.filter(digit => digit%2 == 0 )
const oddArr = arr.filter(digit => digit%2 != 0 )
console.log(evenArr.concat(oddArr))
console.log([...evenArr,...oddArr])
```
